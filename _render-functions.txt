===== initProgramDragAndDrop =====
function initProgramDragAndDrop(host){let dragIndex=null,dragDay='',dragRow=null;host.querySelectorAll('[data-program-drag]').forEach(handle=>{handle.addEventListener('click',e=>{e.preventDefault();e.stopPropagation()});handle.addEventListener('pointerdown',e=>e.stopPropagation());handle.addEventListener('dragstart',e=>{dragIndex=Number(handle.dataset.programDrag);dragDay=handle.dataset.dragDay||'';dragRow=handle.closest('.event-row');if(!dragRow||!Number.isInteger(dragIndex)){e.preventDefault();return}e.dataTransfer.effectAllowed='move';e.dataTransfer.setData('text/plain',String(dragIndex));requestAnimationFrame(()=>dragRow?.classList.add('is-dragging'))});handle.addEventListener('dragend',()=>{host.querySelectorAll('.event-row').forEach(row=>row.classList.remove('is-dragging'));host.querySelectorAll('.events').forEach(box=>box.classList.remove('is-drag-over'));dragIndex=null;dragDay='';dragRow=null})});host.querySelectorAll('.events[data-day]').forEach(container=>{container.addEventListener('dragover',e=>{if(dragIndex===null||container.dataset.day!==dragDay)return;e.preventDefault();e.dataTransfer.dropEffect='move';container.classList.add('is-drag-over');const rows=[...container.querySelectorAll('.event-row:not(.is-dragging)')];const target=rows.find(row=>e.clientY<row.getBoundingClientRect().top+row.getBoundingClientRect().height/2);if(target)container.insertBefore(dragRow,target);else container.appendChild(dragRow)});container.addEventListener('dragleave',e=>{if(!container.contains(e.relatedTarget))container.classList.remove('is-drag-over')});container.addEventListener('drop',e=>{if(dragIndex===null||container.dataset.day!==dragDay)return;e.preventDefault();container.classList.remove('is-drag-over');const visibleOrder=[...container.querySelectorAll('.event-row')].map(row=>Number(row.dataset.index)).filter(Number.isInteger);const visibleSet=new Set(visibleOrder);let cursor=0;const fullOrder=getOrderedProgramRows(dragDay).map(x=>x.i).map(index=>visibleSet.has(index)?visibleOrder[cursor++]:index);resequenceProgramDay(dragDay,fullOrder);save();renderProgram();toast('تم حفظ ترتيب صفوف اليوم')})})}

===== applyAdminMode =====
function applyAdminMode(){
  document.body.classList.toggle("admin-mode",Boolean(isAdmin));
  ["sponsors","followups","finance"].forEach(id=>{const section=document.getElementById(id);if(section){section.hidden=!isAdmin;section.setAttribute("aria-hidden",isAdmin?"false":"true")}})
  const btn=document.getElementById("adminAccessBtn");
  if(btn){btn.textContent=isAdmin?"الخروج من وضع الإدارة":"إدارة البرنامج";btn.setAttribute("aria-pressed",isAdmin?"true":"false")}
  const badge=document.getElementById("adminModeBadge");
  if(badge){badge.hidden=!isAdmin;badge.setAttribute("aria-hidden",isAdmin?"false":"true")}
  renderAll()
}
